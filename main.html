esg_extractor.js 
javascript:(function(){
  const tables = document.querySelectorAll('table');
  let esgData = [];
  tables.forEach(t => {
    if(t.innerText.match(/Scope 1|Emissions|Diversity/i)) {
      const rows = t.querySelectorAll('tr');
      rows.forEach(r => {
        const cells = Array.from(r.querySelectorAll('td, th')).map(c => c.innerText.trim());
        if(cells.length > 0) esgData.push(cells.join('\t'));
      });
    }
  });
  if(esgData.length > 0) {
    navigator.clipboard.writeText(esgData.join('\n'));
    alert('ESG Data copied to clipboard!');
  } else {
    console.log('No relevant ESG metrics found.');
  }
})();
